def check_special_ip(ip):
    a, b, c, d = map(int, ip.split("."))

    if a == 127:
        return "Loopback Address"
    elif ip == "255.255.255.255":
        return "Broadcast Address"
    elif a == 10:
        return "Private Class A Address"
    elif a == 172 and 16 <= b <= 31:
        return "Private Class B Address"
    elif a == 192 and b == 168:
        return "Private Class C Address"
    elif 224 <= a <= 239:
        return "Multicast Address"
    elif a == 169 and b == 254:
        return "Link-local Address"
    else:
        return "Public Address"

def separate_net_host(ip):
    octets = ip.split(".")
    a = int(octets[0])

    if 1 <= a <= 126:
        return octets[0], ".".join(octets[1:])
    elif 128 <= a <= 191:
        return ".".join(octets[:2]), ".".join(octets[2:])
    elif 192 <= a <= 223:
        return ".".join(octets[:3]), octets[3]
    else:
        return "N/A", "N/A"

ip = input("Enter an IP address (e.g., 192.168.1.10): ")

special = check_special_ip(ip)
net_id, host_id = separate_net_host(ip)

print("IP Address:", ip)
print("Special Type:", special)
print("Network ID:", net_id)
print("Host ID:", host_id)

import socket
import threading
import time

# ---------------- SERVER ---------------- #
def server_program():
    server = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    server.bind(("127.0.0.1", 6000))

    print("UDP Server Started...")
    print("------------------------")

    data, addr = server.recvfrom(1024)
    name = data.decode()

    data, _ = server.recvfrom(1024)
    m1 = int(data.decode())

    data, _ = server.recvfrom(1024)
    m2 = int(data.decode())

    data, _ = server.recvfrom(1024)
    m3 = int(data.decode())

    total = m1 + m2 + m3
    average = total / 3

    result = f"""
Student Name : {name}

Mark 1 : {m1}
Mark 2 : {m2}
Mark 3 : {m3}

Total : {total}
Average : {average}
"""

    print(result)
    server.sendto(result.encode(), addr)
    server.close()


# ---------------- CLIENT ---------------- #
def client_program():
    client = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    address = ("127.0.0.1", 6000)

    client.sendto("Sakthi".encode(), address)
    client.sendto("85".encode(), address)
    client.sendto("90".encode(), address)
    client.sendto("95".encode(), address)

    data, _ = client.recvfrom(1024)

    print("Result Received From Server")
    print(data.decode())

    client.close()


# ---------------- MAIN ---------------- #
t1 = threading.Thread(target=server_program)
t1.start()

time.sleep(1)
client_program()

t1.join()

import socket

    ipv4 = socket.gethostbyname(hostname)
    print("IPv4 Address:", ipv4)
except:
    print("IPv4 Address: Not found")

    ipv6_info = socket.getaddrinfo(hostname, None, socket.AF_INET6)

except:
    print("IPv6 Address: Not found")

INFINITY = 16

topology = {
    'A': {'B': 1, 'D': 1},
    'B': {'A': 1, 'C': 1, 'E': 1},
    'C': {'B': 1, 'F': 1},
    'D': {'A': 1, 'E': 1},
    'E': {'B': 1, 'D': 1, 'F': 1},
    'F': {'C': 1, 'E': 1}
}

rt = {
    r: {
        d: [topology[r].get(d, INFINITY),
            d if d in topology[r] else None]
        for d in topology
    }
    for r in topology
}

def update():
    changed = False

    for r in topology:
        for n, c in topology[r].items():
            for d, (dist, nxt) in rt[n].items():
                if rt[r][d][0] > c + dist:
                    rt[r][d] = [c + dist, n]
                    changed = True

    return changed

while update():
    pass

for r in rt:
    print(f"\nRouter {r}")

    for d, (dist, nxt) in rt[r].items():
        print(f"  dist={dist}, next={nxt}")

network = {
    "A": ["B", "C"],
    "B": ["A", "D"],
    "C": ["A"],
    "D": ["B", "E"],
    "E": ["D"]
}

def route_request(src, dst):
    visited = set()
    queue = [[src]]

    while queue:
        path = queue.pop(0)
        node = path[-1]

        if node == dst:
            return path

        if node not in visited:
            for neighbor in network.get(node, []):
                queue.append(path + [neighbor])
            visited.add(node)

    return None

src, dst = "A", "E"

print(f"Source: {src},Destination: {dst}")

route = route_request(src, dst)

if route:
    print("RREQ Broadcasted...")
    print("RREP Received....")
    print("Route Established:", "<-".join(route))
else:
    print("No Route Found")

inf = 999

cost = [
    [0, 2, inf, 1],
    [2, 0, 3, 2],
    [inf, 3, 0, 4],
    [1, 2, 4, 0]
]

n = len(cost)
dist = [row[:] for row in cost]

for k in range(n):
    for i in range(n):
        for j in range(n):
            if dist[i][j] > dist[i][k] + dist[k][j]:
                dist[i][j] = dist[i][k] + dist[k][j]

print("Distance Vector:")

for i in range(n):
    print(f"From node {i}: {dist[i]}")
